DECLARE @v_QUERY1 as varchar(max)='';

DECLARE @v_QUERY2 as varchar(max)='';

		SET @TO_DATE=(SELECT  CAST(MAX(isnull(MT.MODIFY_DATE,MT.CREATED_DATE)) AS VARCHAR) FROM QMS_AUDIT_MT MT)
		PRINT @TO_DATE

SELECT @USER_TYPE = URM.ROLE_NAME FROM USER_MT UM
         INNER JOIN USER_ROLE_MT URM ON UM.ROLE_ID = URM.ROLE_ID  
         WHERE USER_ID = @user_id

CREATE TABLE #TEMP_QMS_DASH_MT (QMS_AUDIT_ID INT, VENDOR_NAME VARCHAR(MAX), VENDOR_ID VARCHAR(MAX), SUB_VENDOR VARCHAR(MAX), FACTORY_ID INT
                                 , FINAL_RATING DECIMAL(18, 2), SOURCING_MERCHANT_ID INT, QA_ID INT, GRADE VARCHAR(MAX),BADGE VARCHAR(20),BG_COLOR NVARCHAR(20),
								 CATEGORY VARCHAR(MAX),EXACT_PERCENTAGE VARCHAR(MAX), AUDIT_DATE VARCHAR(MAX), AUDIT_DUE_DATE VARCHAR(MAX))

  CREATE TABLE #TEMP_STYLE_MT (VENDOR_ID INT ,FACTORY_ID INT)
  DECLARE @FILTER_VALUE INT=0                               
  SET @V_QUERY3=' SELECT DISTINCT FG_VENDOR_ID,FACTORY_ID FROM STYLE_MT SMT WITH(NOLOCK) where 1=1  ' 

	IF EXISTS(SELECT 1 FROM  USER_CHART_FILTER_DT  WITH (NOLOCK) WHERE CHART_CODE = 'QMS' AND UPPER(FILTER_TYPE) = 'Brand' AND USER_ID = @user_id) 
	BEGIN
		SET @FILTER_VALUE =1
		SET @V_QUERY3=@V_QUERY3+' AND BRAND_ID IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT  WITH (NOLOCK)   
									WHERE UPPER(FILTER_TYPE)=''Brand'' AND CHART_CODE=''QMS'' AND USER_ID='+ CAST(@user_id AS VARCHAR(4))+') '
	END

	IF EXISTS(SELECT 1 FROM  USER_CHART_FILTER_DT  WITH (NOLOCK) WHERE CHART_CODE = 'QMS' AND UPPER(FILTER_TYPE) = 'CATEGORY' AND USER_ID = @user_id) 
	BEGIN
		SET @FILTER_VALUE =1 
		SET @V_QUERY3=@V_QUERY3+' AND CATEGORY_ID IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT  WITH (NOLOCK)   
		WHERE UPPER(FILTER_TYPE)=''CATEGORY'' AND CHART_CODE=''QMS'' AND USER_ID='+ CAST(@user_id AS VARCHAR(4))+') '
	END

	PRINT @V_QUERY3
	INSERT INTO #TEMP_STYLE_MT(VENDOR_ID,FACTORY_ID)                              
	EXECUTE (@V_QUERY3)
	
 SET @V_QUERY3=''
 IF EXISTS(SELECT 1 FROM  USER_CHART_FILTER_DT  WITH (NOLOCK) WHERE CHART_CODE = 'QMS' AND UPPER(FILTER_TYPE) = 'SVendor' AND USER_ID = @user_id)  
	SET @V_QUERY3=@V_QUERY3+' AND MT.FACTORY_ID IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT  WITH (NOLOCK)   
	WHERE UPPER(FILTER_TYPE)=''SVendor'' AND CHART_CODE=''QMS'' AND USER_ID='+ CAST(@user_id AS VARCHAR(4))+') '
 
	IF EXISTS(SELECT 1 FROM  USER_CHART_FILTER_DT  WITH (NOLOCK) WHERE CHART_CODE = 'QMS' AND UPPER(FILTER_TYPE) = 'Vendor' AND USER_ID = @user_id)  
	SET @V_QUERY3=@V_QUERY3+' AND MT.VENDOR_ID IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT  WITH (NOLOCK)   
	WHERE UPPER(FILTER_TYPE)=''Vendor'' AND CHART_CODE=''QMS'' AND USER_ID='+ CAST(@user_id AS VARCHAR(4))+') '
 
 IF EXISTS(SELECT 1 FROM  USER_CHART_FILTER_DT  WITH (NOLOCK) WHERE CHART_CODE = 'QMS' AND UPPER(FILTER_TYPE) = 'QA' AND USER_ID = @user_id)  
	SET @V_QUERY3=@V_QUERY3+' AND CREATED_BY IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT  WITH (NOLOCK)   
	WHERE UPPER(FILTER_TYPE)=''QA'' AND CHART_CODE=''QMS'' AND USER_ID='+ CAST(@user_id AS VARCHAR(4))+') '
                                                                                 
	SET @v_QUERY1 = ' SELECT DISTINCT QMS_AUDIT_ID
	, (SELECT ISNULL(VENDOR_NAME, '''') FROM VENDOR_MT WITH(NOLOCK) WHERE VENDOR_ID = MT.VENDOR_ID)
	, MT.VENDOR_ID
	, (SELECT ISNULL(VENDOR_FACTORY, '''') FROM VENDOR_FACTORY_DT WITH(NOLOCK) WHERE VENDOR_FACTORY_ID = MT.FACTORY_ID)
	, MT.FACTORY_ID
	, isnull(RESULT_PER,0)
	--, (SELECT CAST(((ACTUAL_SCORED * 100) / TOTAL_MAX_SCORE) AS DECIMAL(18, 2)) FROM #TEMP_ACTUAL_SCORE WITH(NOLOCK) WHERE QMS_AUDIT_ID = MT.QMS_AUDIT_ID)
	, (SELECT TOP 1 ISNULL(SOURCING_MERCHANT_ID, 0) FROM STYLE_MT WITH(NOLOCK) WHERE FG_VENDOR_ID = MT.VENDOR_ID AND FACTORY_ID = MT.FACTORY_ID)
	, (SELECT TOP 1 ISNULL(QUALITY_MANAGER_ID, 0) FROM STYLE_MT WITH(NOLOCK) WHERE FG_VENDOR_ID = MT.VENDOR_ID AND FACTORY_ID = MT.FACTORY_ID)
	,FORMAT(isnull(MT.MODIFY_DATE,MT.CREATED_DATE),''dd-MMM-yyyy'') AS AUDIT_DATE
	,FORMAT(isnull(MT.MODIFY_DATE,MT.CREATED_DATE),''dd-MMM-yyyy'') AS AUDIT_DUE_DATE
	FROM QMS_AUDIT_MT MT WITH(NOLOCK) '
	IF(ISNULL(@FILTER_VALUE,0)=1)
	BEGIN
		SET @v_QUERY1= @v_QUERY1+ ' INNER JOIN #TEMP_STYLE_MT TEMP ON MT.VENDOR_ID=TEMP.VENDOR_ID AND MT.FACTORY_ID=TEMP.FACTORY_ID '
	END
	SET @v_QUERY1 = @v_QUERY1+ ' WHERE MT.FACTORY_ID IN (SELECT DISTINCT VENDOR_FACTORY_ID FROM VENDOR_FACTORY_DT WITH(NOLOCK))  AND SUBMITTED_STATUS=1 ' --(ONLY AUDIT SUBMITTED RESULT HAVE TO SHOW)

	SET @v_QUERY1 = @v_QUERY1+ ' '+ @V_QUERY3 +' '

--if exists(select 1 from USER_CHART_FILTER_DT where CHART_CODE='DSY' AND UPPER(FILTER_TYPE)='SEASON' AND USER_ID=@user_id)

--SET @v_QUERY1=@v_QUERY1+' WHERE SM.SEASON_ID IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT WHERE UPPER(filter_type)=''GBLSEASON'' AND CHART_CODE=''GBL'' AND USER_ID=0) '
		

IF EXISTS(SELECT 1 FROM  USER_CHART_FILTER_DT  WITH (NOLOCK) WHERE CHART_CODE = 'QMS' AND UPPER(FILTER_TYPE) = 'Result' AND USER_ID = @user_id)  
SET @V_QUERY4=@V_QUERY4+' AND CATEGORY IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT  WITH (NOLOCK)   
 WHERE UPPER(FILTER_TYPE)=''Result'' AND CHART_CODE=''QMS'' AND USER_ID='+ CAST(@user_id AS VARCHAR(4))+') '
 
IF EXISTS(SELECT 1 FROM  USER_CHART_FILTER_DT  WITH (NOLOCK) WHERE CHART_CODE = 'QMS' AND UPPER(FILTER_TYPE) = 'FRating' AND USER_ID = @user_id)  
SET @V_QUERY4=@V_QUERY4+' AND  GRADE IN (SELECT FILTER_VALUE FROM USER_CHART_FILTER_DT  WITH (NOLOCK)   
 WHERE UPPER(FILTER_TYPE)=''FRating'' AND CHART_CODE=''QMS'' AND USER_ID='+ CAST(@user_id AS VARCHAR(4))+') '
           
SET @V_QUERY4 = @V_QUERY4+ ' AND  CAST(AUDIT_DATE  AS DATE) BETWEEN CAST('''+ CAST(@FROM_DATE AS VARCHAR)+'''  AS  DATE) AND CAST('''+ CAST(@TO_DATE AS VARCHAR)+''' AS  DATE) '


	INSERT INTO #TEMP_QMS_DASH_MT(QMS_AUDIT_ID, VENDOR_NAME, VENDOR_ID, SUB_VENDOR, FACTORY_ID
	, FINAL_RATING, SOURCING_MERCHANT_ID, QA_ID, AUDIT_DATE , AUDIT_DUE_DATE)

	EXECUTE (@v_QUERY1)
	print (@v_QUERY1)
	
UPDATE MT SET GRADE = CASE WHEN FINAL_RATING BETWEEN 0 AND 50.99 THEN 'D'
                                        WHEN FINAL_RATING BETWEEN 51 AND 65.99 THEN 'C' 
                                        WHEN FINAL_RATING BETWEEN 66 AND 80.99 THEN 'B'
                                        WHEN FINAL_RATING BETWEEN 81 AND 100 THEN 'A' END,

            BADGE = CASE WHEN FINAL_RATING BETWEEN 0 AND 50.99 THEN 'Red'
                                        WHEN FINAL_RATING BETWEEN 51 AND 65.99 THEN 'Yellow' 
                                        WHEN FINAL_RATING BETWEEN 66 AND 80.99 THEN 'Bronze'
                                        WHEN FINAL_RATING BETWEEN 81 AND 100 THEN 'Silver' END,
										
            BG_COLOR = CASE WHEN FINAL_RATING BETWEEN 0 AND 50.99 THEN 'Red'
                                        WHEN FINAL_RATING BETWEEN 51 AND 65.99 THEN 'Yellow' 
                                        WHEN FINAL_RATING BETWEEN 66 AND 80.99 THEN '#CD7F32'
                                        WHEN FINAL_RATING BETWEEN 81 AND 100 THEN 'Silver' END,
										
           CATEGORY = CASE WHEN FINAL_RATING BETWEEN 0 AND 50.99 THEN 'Unsuccessful' 
                              WHEN FINAL_RATING BETWEEN 51 AND 65.99 THEN 'Inconsistent' 
                              WHEN FINAL_RATING BETWEEN 66 AND 80.99 THEN 'Meet Minimum Standards' 
                              WHEN FINAL_RATING BETWEEN 81 AND 100 THEN 'Consistemtly Exceeds Minimum Standards' END ,

			EXACT_PERCENTAGE = FINAL_RATING ,

			AUDIT_DUE_DATE = CASE WHEN FINAL_RATING BETWEEN 0 AND 50.99 THEN FORMAT(DATEADD(DAY,0,AUDIT_DUE_DATE) ,'dd-MMM-yyyy')
                              WHEN FINAL_RATING BETWEEN 51 AND 65.99 THEN FORMAT(DATEADD(DAY,182,AUDIT_DUE_DATE) ,'dd-MMM-yyyy')
                              WHEN FINAL_RATING BETWEEN 66 AND 80.99 THEN FORMAT(DATEADD(DAY,365,AUDIT_DUE_DATE) ,'dd-MMM-yyyy')
                              WHEN FINAL_RATING BETWEEN 81 AND 100 THEN FORMAT(DATEADD(DAY,547,AUDIT_DUE_DATE) ,'dd-MMM-yyyy') END 
FROM #TEMP_QMS_DASH_MT MT WITH (NOLOCK)
		
SET @V_QUERY2 = ' SELECT VENDOR_NAME, SUB_VENDOR, CATEGORY AS RESULT, GRADE AS FINAL_RATING, BADGE, AUDIT_DUE_DATE AS VALIDITY_DATE, EXACT_PERCENTAGE, AUDIT_DATE FROM #TEMP_QMS_DASH_MT MT  WHERE 1=1  '+ @V_QUERY4 +' '
                      
            EXECUTE (@V_QUERY2)
			PRINT (@V_QUERY2)
